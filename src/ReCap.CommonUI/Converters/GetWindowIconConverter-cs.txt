using System;
using System.Diagnostics;
using System.Globalization;
using System.IO;
using Avalonia;
using Avalonia.Controls;
using Avalonia.Data.Converters;
using Avalonia.Media;
using Avalonia.Media.Imaging;
using ReCap.CommonUI.Util;
using ReCap.CommonUI.Util.Reflection;
using ReCap.CommonUI.Util.Win32;

namespace ReCap.CommonUI.Converters
{
    public sealed class GetWindowIconConverter
        : IValueConverter
    {
        public static readonly GetWindowIconConverter Instance = new();
        private GetWindowIconConverter()
        {}




        public object Convert(object value, Type targetType, object parameter, CultureInfo culture)
        {
            Bitmap bitmap;
            if (value is Window window)
            {
                var icon = window.Icon;
                if (icon != null)
                {
                    using MemoryStream iconStream = new();
                    icon.Save(iconStream);
                    iconStream.Position = 0;
                    bitmap = new(iconStream);
                    return GetFromBitmap(bitmap, targetType);
                }
            }

            if (OSInfo.IsWindows && TryGetWindowsGetAppIcon(out bitmap))
                return GetFromBitmap(bitmap, targetType);

            return null;
        }

        object GetFromBitmap(Bitmap bitmap, Type targetType)
        {
            if (targetType == null)
                return bitmap;
            else if (targetType.IsAssignableTo(typeof(IBrush)))
                return new ImageBrush(bitmap);
            else
                return bitmap;
        }


        public object ConvertBack(object value, Type targetType, object parameter, CultureInfo culture)
            => throw new NotSupportedException();


        static bool TryGetWindowsGetAppIcon(out Bitmap bitmap)
        {
            string exePath = Process.GetCurrentProcess().MainModule.FileName;
            if (!File.Exists(exePath))
                goto fail;

            Win32Methods.ExtractIconEx(exePath, 0, out _, out IntPtr hIconSmall, 1);
            if (hIconSmall == IntPtr.Zero)
                goto fail;

            using (var icon = System.Drawing.Icon.FromHandle(hIconSmall))
            {
                using MemoryStream iconStream = new();
                icon.Save(iconStream);
                iconStream.Position = 0;
                bitmap = new(iconStream);
                return bitmap != null;
            }

            fail:
            bitmap = null;
            return false;
        }
    }
}
